What changed (owner's requests):
1. **Weapon tree hub.** The weapon tree's perks now carry a `Branch` (Pistol, Rifle, Shotgun). The hub page shows
   the root perk (Heavy Rounds) and one emblem per gun: pseudo-nodes built in `GetSkillNodes` (`bCategory`, Id
   `Branch_<Gun>`, not in the data). Confirming or clicking an emblem opens that gun's page (the cursor moves to the
   branch's first perk); the shown page is always derived from the cursor node's (Tree, Branch). Back (button, Esc,
   gamepad B through `MenuBack`) returns to the hub on that gun's emblem. Requirement links to nodes on another page
   are not drawn; the emblems hang from the hub's root perks.
2. **Data:** the nine perks keep effects, costs and ranks; two prerequisites changed (Rapid Reload now needs Heavy
   Rounds, Arcane Conduit needs Point Blank). Three descriptions now start with "All guns:".
3. **Skill pages take their tree's look for the whole screen** (`GetMenuSkin`): on the Skill Trees screen and the
   inventory's Skills page, weapons = cyberpunk, spells = RPG, whatever the era.
4. **Spell page:** a darker panel background.
5. Grid: the weapon pages are centred vertically; the emblems are 2.3 perk boxes tall.

   ending on a node of a page that is not drawn; Back that closes the whole menu instead of returning (or the

2. Data: any requirement cycle or a perk that can no longer be reached; anything else (code, smoke, the balance tool

3. Layout at 1280x720 and 2560x1440: the emblem's outline glyph, title and rank segments overlapping or leaving the
   box; the Back button and the "WEAPONS // GUN" title overlapping on a narrow panel; the centred grid running out
   of the panel.
4. Per-frame cost: `GetSkillNodes()` builds an array; it is now also called from `GetMenuSkin` and the footer every

Do not suggest style rewrites, renames or new features, and do not judge the design (perk grouping, art). For each

163: int32 AGBHUD::GetShownSkillTree() const
164: {
165: 	const TArray<FSkillNodeRef> Nodes = GetSkillNodes();
166: 	const FSkillNodeRef* Current = Nodes.FindByPredicate([this](const FSkillNodeRef& Node) { return Node.Id == MenuState.SkillCursor; });
167: 	// No cursor yet: the page draws the first node's tree (DrawSkillTrees), so the look follows the same rule.

171: void AGBHUD::SelectSkillTree(int32 Tree)
172: {
173: 	Tree = FMath::Clamp(Tree, 0, 1);
174: 	const TArray<FSkillNodeRef> Nodes = GetSkillNodes();
175: 	const FSkillNodeRef* Current = Nodes.FindByPredicate([this](const FSkillNodeRef& Node) { return Node.Id == MenuState.SkillCursor; });
176: 	if (Current)
177: 	{
178: 		if (Current->Tree == Tree)
179: 		{
180: 			return;
181: 		}
182: 		MenuState.TreeCursor[Current->Tree] = Current->Id;
183: 		// The clickables of the tree being left are stale until the next draw: a hover or a click on them must not
184: 		// pull the cursor back to (or unlock) a node that is no longer shown (Sol 6.1 review).
185: 		MenuHits.Reset();
186: 	}
187: 	const FName Remembered = MenuState.TreeCursor[Tree];
188: 	const FSkillNodeRef* Target = Nodes.FindByPredicate([Remembered, Tree](const FSkillNodeRef& Node) { return Node.Tree == Tree && Node.Id == Remembered; });
189: 	if (!Target)
190: 	{
191: 		// The tree's front page: the weapon tree opens on its hub, not inside a gun branch.
192: 		Target = Nodes.FindByPredicate([Tree](const FSkillNodeRef& Node) { return Node.Tree == Tree && Node.Branch == NAME_None && !Node.bCategory; });
193: 	}
194: 	if (!Target)
195: 	{
196: 		Target = Nodes.FindByPredicate([Tree](const FSkillNodeRef& Node) { return Node.Tree == Tree; });
197: 	}
198: 	if (Target)
199: 	{
200: 		MenuState.SkillCursor = Target->Id;
201: 	}
202: 	else if (!Current && Nodes.Num() > 0)
203: 	{
204: 		MenuState.SkillCursor = Nodes[0].Id;
205: 	}
206: }

249: const AGBHUD::FMenuHit* AGBHUD::FindMenuHit(const FVector2D& CanvasPosition) const
250: {
251: 	// Last drawn is on top.
252: 	for (int32 i = MenuHits.Num() - 1; i >= 0; --i)
253: 	{
254: 		const FMenuHit& Hit = MenuHits[i];
255: 		if (CanvasPosition.X >= Hit.Min.X && CanvasPosition.X <= Hit.Max.X && CanvasPosition.Y >= Hit.Min.Y && CanvasPosition.Y <= Hit.Max.Y)
256: 		{
257: 			return &Hit;
258: 		}
259: 	}
260: 	return nullptr;

263: void AGBHUD::UpdateMenuMouse()
264: {
265: 	APlayerController* PC = GetOwningPlayerController();
266: 	float MouseX = 0.f;
267: 	float MouseY = 0.f;
268: 	if (!PC || !PC->GetMousePosition(MouseX, MouseY))
269: 	{
270: 		return;
271: 	}
272: 	const FVector2D Mouse(MouseX * MouseToCanvas, MouseY * MouseToCanvas);
273: 	const bool bMoved = bHasMenuMouse && !Mouse.Equals(LastMenuMouse, 0.5f);
274: 	LastMenuMouse = Mouse;
275: 	bHasMenuMouse = true;
276: 	// Hover uses last frame's clickables, so the highlight moves in the same frame it is drawn.
277: 	if (bMoved && MenuHitsMenu == GetOpenMenu())
278: 	{
279: 		if (const FMenuHit* Hit = FindMenuHit(Mouse))
280: 		{
281: 			ApplyMenuHover(*Hit);
282: 		}
283: 	}
284: }

286: void AGBHUD::ApplyMenuHover(const FMenuHit& Hit)
287: {
288: 	switch (Hit.Kind)
289: 	{
290: 	case EMenuHitKind::PauseEntry:
291: 	case EMenuHitKind::ItemRow:
292: 	case EMenuHitKind::WeaponRow:
293: 		MenuState.Cursor = Hit.Index;
294: 		break;
295: 	case EMenuHitKind::SpellSlot:
296: 		if (!MenuState.bChoosingSpell)
297: 		{
298: 			MenuState.Cursor = Hit.Index;
299: 		}
300: 		break;
301: 	case EMenuHitKind::SpellChoice:
302: 		MenuState.SubCursor = Hit.Index;
303: 		break;
304: 	case EMenuHitKind::SkillNode:
305: 		MenuState.SkillCursor = Hit.Id;
306: 		break;
307: 	case EMenuHitKind::Tab:
308: 	case EMenuHitKind::SkillReset:
309: 	case EMenuHitKind::SkillAddPoint:
310: 	default:
311: 		// Tabs, the skill buttons (reset, +1 point) and the death screen only react to clicks.
312: 		break;
313: 	}

316: void AGBHUD::ActivateMenuHit(const FMenuHit& Hit)
317: {
318: 	AGBPlayerCharacter* Player = GetPlayer();
319: 	switch (Hit.Kind)
320: 	{
321: 	case EMenuHitKind::PauseEntry:
322: 		MenuState.Cursor = Hit.Index;
323: 		MenuConfirm();
324: 		break;
325: 	case EMenuHitKind::Tab:
326: 		if (Hit.Index != MenuState.Page)
327: 		{
328: 			SetInventoryPage(Hit.Index);
329: 		}
330: 		break;
331: 	case EMenuHitKind::ItemRow:
332: 		MenuState.Cursor = Hit.Index;
333: 		ConfirmItem(Player);
334: 		break;
335: 	case EMenuHitKind::WeaponRow:
336: 		MenuState.Cursor = Hit.Index;
337: 		ConfirmWeapon(Player);
338: 		break;
339: 	case EMenuHitKind::SpellSlot:
340: 		// A click on another slot while choosing leaves the chooser and opens it for that slot.
341: 		MenuState.bChoosingSpell = false;
342: 		MenuState.Cursor = Hit.Index;
343: 		ConfirmSpellSlot(Player);
344: 		break;
345: 	case EMenuHitKind::SpellChoice:
346: 		MenuState.SubCursor = Hit.Index;
347: 		ConfirmSpellSlot(Player);
348: 		break;
349: 	case EMenuHitKind::SkillNode:
350: 		MenuState.SkillCursor = Hit.Id;
351: 		ConfirmSkill(Player);
352: 		break;
353: 	case EMenuHitKind::SkillReset:
354: 		ResetSkills(Player);
355: 		break;
356: 	case EMenuHitKind::SkillAddPoint:
357: 		AddTestSkillPoint(Player);
358: 		break;
359: 	case EMenuHitKind::SkillTreeTab:
360: 		SelectSkillTree(Hit.Index);
361: 		break;
362: 	case EMenuHitKind::SkillBranchBack:
363: 		CloseSkillBranch();
364: 		break;
365: 	case EMenuHitKind::DeathScreen:
366: 		// The controller handles the respawn confirm (AGBPlayerController::InputMenuClick): nothing here,
367: 		// so a click respawns exactly once.
368: 		break;
369: 	}
370: }

372: bool AGBHUD::MenuClick(const FVector2D& ScreenPosition)
373: {
374: 	const EGBMenu Menu = GetOpenMenu();
375: 	// Only the clickables of the screen that is showing right now (not of a page left this frame).
376: 	if (Menu == EGBMenu::None || Menu != MenuHitsMenu || MenuHitsPage != MenuState.Page || bMenuHitsChoosing != MenuState.bChoosingSpell)
377: 	{
378: 		return false;
379: 	}
380: 	const FVector2D Position = ScreenPosition * MouseToCanvas;
381: 	const FMenuHit* Found = FindMenuHit(Position);
382: 	if (!Found)
383: 	{
384: 		return false;
385: 	}
386: 	// Copy: the action may change the menu, and the list is rebuilt on the next draw.
387: 	const FMenuHit Hit = *Found;
388: 	ActivateMenuHit(Hit);
389: 	return true;
390: }

392: void AGBHUD::MenuNavigate(FIntPoint Direction)
393: {
394: 	using namespace GBHudMenuLocal;
395: 	AGBPlayerCharacter* Player = GetPlayer();
396: 	switch (GetOpenMenu())
397: 	{
398: 	case EGBMenu::Pause:
399: 		if (Direction.Y != 0)
400: 		{
401: 			MenuState.Cursor = Wrap(MenuState.Cursor + Direction.Y, 2);
402: 		}
403: 		break;
404: 	case EGBMenu::SkillTrees:
405: 		NavigateSkills(Direction);
406: 		break;
407: 	case EGBMenu::Inventory:
408: 		if (MenuState.Page == PageSkills)
409: 		{
410: 			NavigateSkills(Direction);
411: 		}
412: 		else if (MenuState.Page == PageSpells)
413: 		{
414: 			if (Direction.Y == 0)
415: 			{
416: 				break;
417: 			}
418: 			if (MenuState.bChoosingSpell)
419: 			{
420: 				MenuState.SubCursor = Wrap(MenuState.SubCursor + Direction.Y, GetAssignableSpells(Player).Num());
421: 			}
422: 			else
423: 			{
424: 				MenuState.Cursor = Wrap(MenuState.Cursor + Direction.Y, UGBSpellbookComponent::NumSlots + 1);
425: 			}
426: 		}
427: 		else if (MenuState.Page == PageWeapons)
428: 		{
429: 			if (Direction.Y != 0)
430: 			{
431: 				MenuState.Cursor = Wrap(MenuState.Cursor + Direction.Y, GetWeaponList(Player).Num());
432: 			}
433: 		}
434: 		else if (Direction.Y != 0)
435: 		{
436: 			MenuState.Cursor = Wrap(MenuState.Cursor + Direction.Y, GetInventoryList(Player).Num());
437: 		}
438: 		break;
439: 	default:
440: 		break;
441: 	}
442: }

478: 		else

480: 			ConfirmSkill(Player);

489: bool AGBHUD::MenuBack()
490: {
491: 	switch (GetOpenMenu())
492: 	{
493: 	case EGBMenu::Inventory:
494: 		if (MenuState.Page == PageItems && MenuState.PendingCombine != NAME_None)
495: 		{
496: 			MenuState.PendingCombine = NAME_None;
497: 			return true;
498: 		}
499: 		if (MenuState.Page == PageSpells && MenuState.bChoosingSpell)
500: 		{
501: 			MenuState.bChoosingSpell = false;
502: 			return true;
503: 		}
504: 		// A gun branch page goes back to the weapon hub first.
505: 		if (MenuState.Page == PageSkills && CloseSkillBranch())
506: 		{
507: 			return true;
508: 		}
509: 		return false;
510: 	case EGBMenu::SkillTrees:
511: 		return CloseSkillBranch();
512: 	case EGBMenu::Death:
513: 		// The death screen only closes by respawning.
514: 		return true;
515: 	default:
516: 		return false;
517: 	}

520: void AGBHUD::MenuTab(int32 Direction)
521: {
522: 	if (Direction == 0)
523: 	{
524: 		return;
525: 	}
526: 	const EGBMenu Menu = GetOpenMenu();
527: 	if (Menu == EGBMenu::Inventory)
528: 	{
529: 		SetInventoryPage(MenuState.Page + Direction);
530: 	}
531: 	else if (Menu == EGBMenu::SkillTrees)
532: 	{
533: 		// Two pages: either tab key (Q / E, L1 / L2 / R1) turns to the other tree.
534: 		SelectSkillTree(1 - GetShownSkillTree());
535: 	}
610: 	}
611: 	for (const FName& SpellId : Spellbook->GetKnownSpells())
612: 	{
613: 		const FGBSpellDefinition* Def = Data->FindSpell(SpellId);
614: 		if (Def && !Def->bPassive && !Def->bRequiresGun && SpellId != Spellbook->GetEnchantSpell())
615: 		{
616: 			Result.Add(SpellId);
617: 		}
618: 	}
619: 	return Result;
620: }
621: 
622: TArray<AGBHUD::FSkillNodeRef> AGBHUD::GetSkillNodes() const
623: {
624: 	TArray<FSkillNodeRef> Result;
625: 	if (const UGBDataSubsystem* Data = UGBDataSubsystem::Get(this))
626: 	{
627: 		for (const FGBSkillNodeDefinition& Def : Data->GetSkillNodes())
628: 		{
629: 			FSkillNodeRef Ref;
630: 			Ref.Id = Def.Id;
631: 			Ref.Tree = GBHudMenuLocal::IsWeaponTree(Def.Tree) ? 1 : 0;
632: 			Ref.Column = FMath::Max(0, Def.Column);
633: 			Ref.Row = FMath::Max(0, Def.Row);
634: 			Ref.Branch = Ref.Tree == 1 ? Def.Branch : NAME_None;
635: 			Result.Add(Ref);
636: 		}
637: 		// The weapon hub (owner, Nov 13, "similar to Cyberpunk"): the root perks on top, then one emblem per gun
638: 		// branch on the row under them. Known guns first in a fixed order, any other branch after them.
639: 		TArray<FName> Branches = { FName(TEXT("Pistol")), FName(TEXT("Rifle")), FName(TEXT("Shotgun")) };
640: 		Branches.RemoveAll([&Result](FName Branch) { return !Result.ContainsByPredicate([Branch](const FSkillNodeRef& Ref) { return Ref.Branch == Branch; }); });
641: 		int32 HubRows = 0;
642: 		for (const FSkillNodeRef& Ref : Result)
643: 		{
644: 			if (Ref.Tree == 1)
645: 			{
646: 				if (Ref.Branch != NAME_None)
647: 				{
648: 					Branches.AddUnique(Ref.Branch);
649: 				}
650: 				else
651: 				{
652: 					HubRows = FMath::Max(HubRows, Ref.Row + 1);
653: 				}
654: 			}
655: 		}
656: 		for (int32 i = 0; i < Branches.Num(); ++i)
657: 		{
658: 			FSkillNodeRef Category;
659: 			Category.Id = FName(*(TEXT("Branch_") + Branches[i].ToString()));
660: 			Category.Tree = 1;
661: 			Category.Column = i;
662: 			Category.Row = HubRows;
663: 			Category.bCategory = true;
664: 			Category.OpensBranch = Branches[i];
665: 			Result.Add(Category);
666: 		}
667: 	}
668: 	return Result;
669: }
670: 
671: void AGBHUD::OpenSkillBranch(FName Branch)
672: {
673: 	const TArray<FSkillNodeRef> Nodes = GetSkillNodes();
674: 	const FSkillNodeRef* First = nullptr;
675: 	for (const FSkillNodeRef& Node : Nodes)

709: void AGBHUD::NavigateSkills(FIntPoint Direction)
710: {
711: 	const TArray<FSkillNodeRef> Nodes = GetSkillNodes();
712: 	if (Nodes.Num() == 0 || Direction == FIntPoint::ZeroValue)
713: 	{
714: 		return;
715: 	}
716: 	const FSkillNodeRef* Current = Nodes.FindByPredicate([this](const FSkillNodeRef& Node) { return Node.Id == MenuState.SkillCursor; });
717: 	if (!Current)
718: 	{
719: 		MenuState.SkillCursor = Nodes[0].Id;
720: 		return;
721: 	}
722: 
723: 	// One tree per page: pick the nearest node of the shown tree in the pressed direction.
724: 	auto GridPos = [](const FSkillNodeRef& Node)
725: 	{
726: 		return FVector2D(Node.Column, Node.Row);
727: 	};
728: 	const FVector2D Dir(Direction.X, Direction.Y);
729: 	const FVector2D From = GridPos(*Current);
730: 	float BestCost = TNumericLimits<float>::Max();
731: 	FName Best = NAME_None;
732: 	for (const FSkillNodeRef& Node : Nodes)
733: 	{
734: 		if (Node.Id == Current->Id || Node.Tree != Current->Tree || Node.Branch != Current->Branch)
735: 		{
736: 			continue;
737: 		}
738: 		const FVector2D Delta = GridPos(Node) - From;
739: 		const float Along = FVector2D::DotProduct(Delta, Dir);
740: 		if (Along <= 0.f)
741: 		{
742: 			continue;
743: 		}
744: 		const float Across = FMath::Abs(FVector2D::CrossProduct(Delta, Dir));
745: 		const float Cost = Along + 2.f * Across;
746: 		if (Cost < BestCost)
747: 		{
748: 			BestCost = Cost;
749: 			Best = Node.Id;
750: 		}
751: 	}
752: 	if (Best != NAME_None)
753: 	{
754: 		MenuState.SkillCursor = Best;
755: 	}
756: 	else if (Direction.X != 0 && GetOpenMenu() == EGBMenu::Inventory)
757: 	{
758: 		// The inventory's Q / E switch its pages, so there left / right past the tree's edge turns to the other tree
759: 		// (spells on the left, weapons on the right).
760: 		const int32 Other = Direction.X > 0 ? 1 : 0;
761: 		if (Other != Current->Tree)
762: 		{
763: 			SelectSkillTree(Other);
764: 		}
765: 	}
766: }

UGBSkillTreeComponent::UGBSkillTreeComponent()

	PrimaryComponentTick.bCanEverTick = false;

void UGBSkillTreeComponent::BeginPlay()
{
	Super::BeginPlay();

	if (UGBDataSubsystem* Data = UGBDataSubsystem::Get(this))
	{
		DataReloadedHandle = Data->OnDataReloaded.AddUObject(this, &UGBSkillTreeComponent::HandleDataReloaded);
	}
}

void UGBSkillTreeComponent::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
	if (UGBDataSubsystem* Data = UGBDataSubsystem::Get(this))
	{
		Data->OnDataReloaded.Remove(DataReloadedHandle);
	}
	DataReloadedHandle.Reset();

	Super::EndPlay(EndPlayReason);
}

void UGBSkillTreeComponent::InitializeFromData()
{
	RemoveAllModifiers();
	Ranks.Reset();
	PointsSpent = 0;

	const UGBDataSubsystem* Data = UGBDataSubsystem::Get(this);
	if (!Data)
	{
		UE_LOG(LogGB, Warning, TEXT("Skill tree: no data subsystem, starting with 0 points."));
	}
	Points = Data ? FMath::Max(0, Data->GetPlayer().StartingSkillPoints) : 0;
	OnSkillsChanged.Broadcast();
}

void UGBSkillTreeComponent::AddPoints(int32 Amount)
{
	if (Amount == 0)
	{
		return;
	}
	Points = FMath::Max(0, Points + Amount);
	OnSkillsChanged.Broadcast();
}

int32 UGBSkillTreeComponent::GetRank(FName NodeId) const
{
	return Ranks.FindRef(NodeId);
}

EGBSkillUnlockResult UGBSkillTreeComponent::CanUnlock(FName NodeId) const
{
	const UGBDataSubsystem* Data = UGBDataSubsystem::Get(this);
	const FGBSkillNodeDefinition* Node = Data ? Data->FindSkillNode(NodeId) : nullptr;
	if (!Node)
	{
		return EGBSkillUnlockResult::UnknownNode;
	}
	return EvaluateUnlock(*Node, Ranks, Points);
}

EGBSkillUnlockResult UGBSkillTreeComponent::Unlock(FName NodeId)
{
	const UGBDataSubsystem* Data = UGBDataSubsystem::Get(this);
	const FGBSkillNodeDefinition* Node = Data ? Data->FindSkillNode(NodeId) : nullptr;
	if (!Node)
	{
		UE_LOG(LogGB, Warning, TEXT("Unlock: unknown skill node '%s'."), *NodeId.ToString());
		return EGBSkillUnlockResult::UnknownNode;
	}

	const EGBSkillUnlockResult Result = EvaluateUnlock(*Node, Ranks, Points);
	if (Result != EGBSkillUnlockResult::Ok)
	{
		return Result;
	}

	Points -= FMath::Max(0, Node->Cost);
	PointsSpent += FMath::Max(0, Node->Cost);
	Ranks.FindOrAdd(NodeId) += 1;
	ApplyNodeModifiers(NodeId);
	OnSkillsChanged.Broadcast();
	return EGBSkillUnlockResult::Ok;
}

int32 UGBSkillTreeComponent::ResetAll()
{
	if (Ranks.Num() == 0)
	{
		return 0;
	}
	RemoveAllModifiers();
	const int32 Refunded = PointsSpent;
	Points += PointsSpent;
	PointsSpent = 0;
	Ranks.Reset();
	OnSkillsChanged.Broadcast();
	return Refunded;
}

EGBSkillUnlockResult UGBSkillTreeComponent::EvaluateUnlock(const FGBSkillNodeDefinition& Node, const TMap<FName, int32>& InRanks, int32 AvailablePoints)
{
	if (InRanks.FindRef(Node.Id) >= Node.MaxRank)
	{
		return EGBSkillUnlockResult::MaxRank;
	}
	for (const FName& Required : Node.Requires)
	{
		if (!Required.IsNone() && InRanks.FindRef(Required) <= 0)
		{
			return EGBSkillUnlockResult::MissingRequirement;
		}
	}
	if (AvailablePoints < Node.Cost)
	{
		return EGBSkillUnlockResult::NotEnoughPoints;
	}
	return EGBSkillUnlockResult::Ok;
}

void UGBSkillTreeComponent::ApplyNodeModifiers(FName NodeId)
{
	UGBStatsComponent* Stats = UGBStatsComponent::Find(GetOwner());
	if (!Stats)
	{
		return;
	}

	const FName Source = ModifierSource(NodeId);
	const int32 Rank = GetRank(NodeId);
	const UGBDataSubsystem* Data = UGBDataSubsystem::Get(this);
	const FGBSkillNodeDefinition* Node = Data ? Data->FindSkillNode(NodeId) : nullptr;
	if (!Node || Rank <= 0)
	{
		if (!Node && Rank > 0)
		{
			UE_LOG(LogGB, Warning, TEXT("Skill node '%s' is owned but missing from Skills.json; its modifiers are off."), *NodeId.ToString());
		}
		Stats->RemoveModifiers(Source);
		return;
	}

	// ValueScale = rank: Add modifiers count once per rank, Multiply modifiers stack linearly (not compounding).
	Stats->AddModifiers(Source, Node->ModifiersPerRank, static_cast<float>(Rank));
}

void UGBSkillTreeComponent::RemoveAllModifiers()
{
	UGBStatsComponent* Stats = UGBStatsComponent::Find(GetOwner());
	if (!Stats)
	{
		return;
	}
	for (const TPair<FName, int32>& Pair : Ranks)
	{
		Stats->RemoveModifiers(ModifierSource(Pair.Key));
	}
}

void UGBSkillTreeComponent::ReapplyAllModifiers()
{
	for (const TPair<FName, int32>& Pair : Ranks)
	{
		ApplyNodeModifiers(Pair.Key);
	}
}

function applyMods(base, keys, ranks) {
	let add = 0;
	let mult = 1;

		const node = skillById[id];
		if (!node) { throw new Error('unknown skill ' + id); }

			if (m.Op === 'Add') { add += m.Value * rank; }

	/** Reads and parses one file of the data folder. False (and an error) if it is missing or not valid JSON. */
	bool ReadJsonFile(const FString& FileName, TSharedPtr<FJsonObject>& OutRoot)
	{
		const FString Path = UGBDataSubsystem::GetDataDir() / FileName;
		FString Text;
		if (!FFileHelper::LoadFileToString(Text, *Path))
		{
			ReportError(FString::Printf(TEXT("Data: cannot read %s"), *Path));
			return false;
		}

		const TSharedRef<TJsonReader<>> Reader = TJsonReaderFactory<>::Create(Text);
		if (!FJsonSerializer::Deserialize(Reader, OutRoot) || !OutRoot.IsValid())
		{
			ReportError(FString::Printf(TEXT("Data: %s is not valid JSON: %s"), *FileName, *Reader->GetErrorMessage()));
			OutRoot.Reset();
			return false;
		}
		return true;
	}

	/** Loads the array field Field of FileName into Out. Out is left empty on any error. */
	template <typename T>
	void LoadArray(const FString& FileName, const TCHAR* Field, TArray<T>& Out)
	{
		Out.Reset();

		TSharedPtr<FJsonObject> Root;
		if (!ReadJsonFile(FileName, Root))
		{
			return;
		}

		const TArray<TSharedPtr<FJsonValue>>* JsonArray = nullptr;
		if (!Root->TryGetArrayField(Field, JsonArray) || !JsonArray)
		{
			ReportError(FString::Printf(TEXT("Data: %s has no array \"%s\""), *FileName, Field));
			return;
		}

		FText FailReason;
		if (!FJsonObjectConverter::JsonArrayToUStruct(*JsonArray, &Out, 0, 0, false, &FailReason))
		{
			ReportError(FString::Printf(TEXT("Data: %s \"%s\" could not be read: %s"), *FileName, Field, *FailReason.ToString()));
			Out.Reset();
		}
	}

	/** Loads the whole of FileName into Out. Out is reset to defaults on any error. */
	template <typename T>
	void LoadStruct(const FString& FileName, T& Out)
	{
		Out = T();

		TSharedPtr<FJsonObject> Root;
		if (!ReadJsonFile(FileName, Root))
		{
			return;
		}

		FText FailReason;
		if (!FJsonObjectConverter::JsonObjectToUStruct(Root.ToSharedRef(), &Out, 0, 0, false, &FailReason))
		{
			ReportError(FString::Printf(TEXT("Data: %s could not be read: %s"), *FileName, *FailReason.ToString()));
			Out = T();
		}
	}

	/** Id -> index. Warns about empty and duplicate ids; the first entry with an id wins. */
	template <typename T>
	void BuildIndex(const TArray<T>& Rows, TMap<FName, int32>& OutIndex, const TCHAR* FileName)
	{
		OutIndex.Reset();
		for (int32 i = 0; i < Rows.Num(); ++i)
		{
			const FName Id = Rows[i].Id;
			if (Id.IsNone())
			{
				UE_LOG(LogGB, Warning, TEXT("Data: %s entry %d has an empty Id; it cannot be found by id."), FileName, i);
				continue;
			}
			if (OutIndex.Contains(Id))
			{
				UE_LOG(LogGB, Warning, TEXT("Data: %s has a duplicate Id \"%s\" (entry %d); the first one is used."), FileName, *Id.ToString(), i);
				continue;
			}
			OutIndex.Add(Id, i);
		}
	}

	template <typename T>
	const T* FindIn(const TArray<T>& Rows, const TMap<FName, int32>& Index, FName Id)
	{
		const int32* Found = Index.Find(Id);
		return (Found && Rows.IsValidIndex(*Found)) ? &Rows[*Found] : nullptr;
	}

EGBSkillUnlockResult UGBSkillTreeComponent::Unlock(FName NodeId)
{
	const UGBDataSubsystem* Data = UGBDataSubsystem::Get(this);
	const FGBSkillNodeDefinition* Node = Data ? Data->FindSkillNode(NodeId) : nullptr;
	if (!Node)
	{
		UE_LOG(LogGB, Warning, TEXT("Unlock: unknown skill node '%s'."), *NodeId.ToString());
		return EGBSkillUnlockResult::UnknownNode;
	}

	const EGBSkillUnlockResult Result = EvaluateUnlock(*Node, Ranks, Points);
	if (Result != EGBSkillUnlockResult::Ok)
	{
		return Result;
	}

	Points -= FMath::Max(0, Node->Cost);
	PointsSpent += FMath::Max(0, Node->Cost);
	Ranks.FindOrAdd(NodeId) += 1;
	ApplyNodeModifiers(NodeId);
	OnSkillsChanged.Broadcast();
	return EGBSkillUnlockResult::Ok;
}

int32 UGBSkillTreeComponent::ResetAll()
{
	if (Ranks.Num() == 0)
	{
		return 0;
	}
	RemoveAllModifiers();
	const int32 Refunded = PointsSpent;
	Points += PointsSpent;
	PointsSpent = 0;
	Ranks.Reset();
	OnSkillsChanged.Broadcast();
	return Refunded;
}

EGBSkillUnlockResult UGBSkillTreeComponent::EvaluateUnlock(const FGBSkillNodeDefinition& Node, const TMap<FName, int32>& InRanks, int32 AvailablePoints)
{
	if (InRanks.FindRef(Node.Id) >= Node.MaxRank)
	{
		return EGBSkillUnlockResult::MaxRank;
	}
	for (const FName& Required : Node.Requires)
	{
		if (!Required.IsNone() && InRanks.FindRef(Required) <= 0)
		{
			return EGBSkillUnlockResult::MissingRequirement;
		}
	}
	if (AvailablePoints < Node.Cost)
	{
		return EGBSkillUnlockResult::NotEnoughPoints;
	}
	return EGBSkillUnlockResult::Ok;
}

void UGBSkillTreeComponent::ApplyNodeModifiers(FName NodeId)
{
	UGBStatsComponent* Stats = UGBStatsComponent::Find(GetOwner());
	if (!Stats)
	{
		return;
	}

	const FName Source = ModifierSource(NodeId);
	const int32 Rank = GetRank(NodeId);
	const UGBDataSubsystem* Data = UGBDataSubsystem::Get(this);
	const FGBSkillNodeDefinition* Node = Data ? Data->FindSkillNode(NodeId) : nullptr;
	if (!Node || Rank <= 0)
	{
		if (!Node && Rank > 0)
		{
			UE_LOG(LogGB, Warning, TEXT("Skill node '%s' is owned but missing from Skills.json; its modifiers are off."), *NodeId.ToString());
		}
		Stats->RemoveModifiers(Source);
		return;
	}

	// ValueScale = rank: Add modifiers count once per rank, Multiply modifiers stack linearly (not compounding).
	Stats->AddModifiers(Source, Node->ModifiersPerRank, static_cast<float>(Rank));
}
