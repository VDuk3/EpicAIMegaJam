<#
Build-Jam.ps1 - Compiles the Glyphbound C++ module (Editor target by default) and prints only what matters.
One build at a time per machine: do not run this while another build, a package or a test run is going.
The Editor must be closed (a full build cannot link while the Editor holds the DLLs).
System.Management.Automation.RemoteException
Usage:
  powershell -File Tools/Jam/Build-Jam.ps1                 # GlyphboundEditor Win64 Development
  powershell -File Tools/Jam/Build-Jam.ps1 -Target Glyphbound   # game target
System.Management.Automation.RemoteException
Output: full log in Glyphbound/Saved/Logs/build_last.log; the console shows errors (first 60) and the result line.
Exit code: 0 = Result: Succeeded.
#>
param(
    [string]$Project = (Join-Path $PSScriptRoot '..\..\Glyphbound\Glyphbound.uproject'),
    [string]$Target = 'GlyphboundEditor',
    [string]$Config = 'Development',
    [string]$EngineDir = 'C:\Program Files\Epic Games\UE_5.9',
    [int]$MaxErrors = 60
)
$Project = (Resolve-Path $Project).Path
$projDir = Split-Path $Project -Parent
$logDir = Join-Path $projDir 'Saved\Logs'
[void][IO.Directory]::CreateDirectory($logDir)
$log = Join-Path $logDir 'build_last.log'
$bat = Join-Path $EngineDir 'Engine\Build\BatchFiles\Build.bat'
$sw = [Diagnostics.Stopwatch]::StartNew()
& cmd.exe /d /c "`"$bat`" $Target Win64 $Config -project=`"$Project`" -waitmutex > `"$log`" 2>&1"
$code = $LASTEXITCODE
$sw.Stop()
$errors = Select-String -Path $log -Pattern '(: error |: fatal error |error LNK|Error: |ERROR: |UHT.*[Ee]rror)' | Select-Object -First $MaxErrors
if ($errors) { $errors | ForEach-Object { $_.Line.Trim() } }
$result = Select-String -Path $log -Pattern '^Result: ' | Select-Object -Last 1
if ($result) { Write-Host $result.Line } else { Write-Host "No 'Result:' line found (exit $code). See $log" }
Write-Host ("Build time: {0:N0} s  (exit {1})  log: {2}" -f $sw.Elapsed.TotalSeconds, $code, $log)
exit $code

<#
Test-Jam.ps1 - Runs the pure-logic automation tests headless and decides pass/fail from the report.
Rules (Devlog/PREJAM_PREP.md, "Lessons from our previous project"):
  - anchored filters: start with ^ and end with a dot; join several with + (one Editor boot);
  - every run writes to its own new folder; the verdict comes from the report, not only the exit code;
  - a run with fewer tests than -MinTests is a failure (guards against a typo in the filter);
  - one build or test run at a time per machine; the Editor must be closed.
System.Management.Automation.RemoteException
Usage:
  powershell -File Tools/Jam/Test-Jam.ps1                              # the whole suite: ^Glyphbound.
  powershell -File Tools/Jam/Test-Jam.ps1 -Filter '^Glyphbound.QTE.'   # only what was touched
  powershell -File Tools/Jam/Test-Jam.ps1 -Filter '^Glyphbound.QTE.+^Glyphbound.Spells.'
Exit code: 0 = every selected test passed and at least MinTests ran.
#>
param(
    [string]$Project = (Join-Path $PSScriptRoot '..\..\Glyphbound\Glyphbound.uproject'),
    [string]$Filter = '^Glyphbound.',
    [int]$MinTests = 1,
    [string]$EngineDir = 'C:\Program Files\Epic Games\UE_5.9',
    [int]$TimeoutSeconds = 600
)
$Project = (Resolve-Path $Project).Path
$projDir = Split-Path $Project -Parent
$stamp = [DateTime]::UtcNow.ToString('yyyyMMdd-HHmmss')
$runDir = Join-Path $projDir "Saved\TestRuns\$stamp"
[void][IO.Directory]::CreateDirectory($runDir)
$log = Join-Path $runDir 'editor.log'
$exe = Join-Path $EngineDir 'Engine\Binaries\Win64\UnrealEditor-Cmd.exe'
$argList = @(
    "`"$Project`"",
    "-ExecCmds=`"Automation RunTests $Filter; Quit`"",
    '-TestExit="Automation Test Queue Empty"',
    "-ReportExportPath=`"$runDir`"",
    "-abslog=`"$log`"",
    '-unattended', '-nopause', '-nosplash', '-NullRHI', '-nosound', '-nocrashreports'
)
$sw = [Diagnostics.Stopwatch]::StartNew()
$p = Start-Process -FilePath $exe -ArgumentList $argList -PassThru -WindowStyle Hidden
if (-not $p.WaitForExit($TimeoutSeconds * 1000)) {
    Write-Host "TIMEOUT after $TimeoutSeconds s. The Editor process (PID $($p.Id)) is still running; log: $log"
    exit 3
}
$sw.Stop()
$index = Join-Path $runDir 'index.json'
if (-not (Test-Path $index)) {
    Write-Host "FAIL: no report was written (exit $($p.ExitCode)). Did the module load? Log: $log"
    Select-String -Path $log -Pattern 'Error:|error:|No automation tests matched' -ErrorAction SilentlyContinue | Select-Object -First 15 | ForEach-Object { $_.Line }
    exit 2
}
$report = Get-Content $index -Raw -Encoding UTF8 | ConvertFrom-Json
$tests = @($report.tests)
$failed = @($tests | Where-Object { $_.state -ne 'Success' })
$testSeconds = 0.0
foreach ($t in $tests) { $testSeconds += [double]$t.duration }
foreach ($t in $failed) {
    Write-Host "FAILED: $($t.fullTestPath) [$($t.state)]"
    foreach ($e in @($t.entries)) { if ($e.event.type -eq 'Error') { Write-Host "   $($e.event.message)" } }
}
$ensures = @(Select-String -Path $log -Pattern 'Ensure condition failed' -ErrorAction SilentlyContinue).Count
Write-Host ("Tests: {0} run, {1} failed, test time {2:N2} s, wall time {3:N0} s, ensures in log: {4}" -f $tests.Count, $failed.Count, $testSeconds, $sw.Elapsed.TotalSeconds, $ensures)
Write-Host "Report: $runDir"
if ($tests.Count -lt $MinTests) { Write-Host "FAIL: only $($tests.Count) tests ran, expected at least $MinTests (check the filter)."; exit 4 }
if ($failed.Count -gt 0) { exit 1 }
exit 0

+bool AGBPlayerCharacter::FindMantleLedge(FGBMantleLedge& Out, bool bFromGround) const
+{
+	UWorld* World = GetWorld();
+	UCharacterMovementComponent* MoveComp = GetCharacterMovement();
+	const UCapsuleComponent* Capsule = GetCapsuleComponent();
+	if (!World || !MoveComp || !Capsule)
+	{
+		return false;
+	}
+
+	const float Radius = Capsule->GetScaledCapsuleRadius();
+	const float HalfHeight = Capsule->GetScaledCapsuleHalfHeight();
+	const FVector ActorLocation = GetActorLocation();
+	const float Feet = ActorLocation.Z - HalfHeight;
+	const float Yaw = Controller ? Controller->GetControlRotation().Yaw : GetActorRotation().Yaw;
+	const FVector Forward = FRotator(0.f, Yaw, 0.f).Vector();
+	const FVector2D Forward2D = FVector2D(Forward.X, Forward.Y).GetSafeNormal();
+
+	const GBMantle::FMantleParams Params = GetMantleParams();
+	const float ProbeDistance = ReadStat(GBStat::MantleProbeDistance, 40.f);
+	const float FacingMinDot = ReadStat(GBStat::MantleFacingMinDot, 0.5f);
+	const float TopInset = ReadStat(GBStat::MantleTopInset, 10.f);
+
+	float JumpRiseCm = 0.f;
+	if (bFromGround)
+	{
+		const float Gravity = FMath::Abs(World->GetGravityZ()) * MoveComp->GravityScale;
+		JumpRiseCm = GBMantle::JumpRise(MoveComp->JumpZVelocity, Gravity);
+	}
+	const float MinHeight = bFromGround ? Params.MinHeight : Params.AirMinHeight;
+	const float MaxHeight = bFromGround ? JumpRiseCm + Params.MaxReach : Params.MaxReach;
+	if (MaxHeight < MinHeight)
+	{
+		return false;
+	}
+
+	// The sweep answers as the capsule would: with its own collision responses, ignoring this pawn.
+	FCollisionQueryParams QueryParams(SCENE_QUERY_STAT(GBMantleWall), false, this);
+	FCollisionResponseParams ResponseParams(Capsule->GetCollisionResponseToChannels());
+
+	// ---- Wall: the shrunk capsule swept forward. ----
+	const float ShrunkRadius = FMath::Max(1.f, Radius - 2.f);
+	const FCollisionShape WallShape = FCollisionShape::MakeCapsule(ShrunkRadius, HalfHeight);
+	FHitResult WallHit;
+	if (!World->SweepSingleByChannel(WallHit, ActorLocation, ActorLocation + Forward * ProbeDistance,
+		FQuat::Identity, ECC_Pawn, WallShape, QueryParams, ResponseParams)
+		|| !WallHit.bBlockingHit || WallHit.bStartPenetrating)
+	{
+		return false;
+	}
+	AActor* WallActor = WallHit.GetActor();
+	if (!WallActor || WallActor->IsA<APawn>())
+	{
+		return false;
+	}
+	if (FMath::Abs(WallHit.ImpactNormal.Z) >= 0.3f)
+	{
+		return false;
+	}
+	FVector2D WallNormal2D(WallHit.ImpactNormal.X, WallHit.ImpactNormal.Y);
+	if (!WallNormal2D.Normalize())
+	{
+		return false;
+	}
+	if (FVector2D::DotProduct(Forward2D, -WallNormal2D) < FacingMinDot)
+	{
+		return false;
+	}
+
+	// ---- Top: a small sphere swept down behind the wall face. ----
+	const FVector2D Edge2D(WallHit.ImpactPoint.X, WallHit.ImpactPoint.Y);
+	const FVector2D TopXY = Edge2D - WallNormal2D * (TopInset + 8.f);
+	const FVector TopStart(TopXY.X, TopXY.Y, Feet + MaxHeight + 20.f);
+	const FVector TopEnd(TopXY.X, TopXY.Y, Feet + MinHeight - 5.f);
+	FHitResult TopHit;
+	if (!World->SweepSingleByChannel(TopHit, TopStart, TopEnd, FQuat::Identity, ECC_Pawn,
+		FCollisionShape::MakeSphere(8.f), QueryParams, ResponseParams)
+		|| !TopHit.bBlockingHit || TopHit.bStartPenetrating)
+	{
+		return false;
+	}
+	AActor* TopActor = TopHit.GetActor();
+	if (!TopActor || TopActor->IsA<APawn>())
+	{
+		return false;
+	}
+	if (TopHit.ImpactNormal.Z < MoveComp->GetWalkableFloorZ())
+	{
+		return false;
+	}
+	const float LedgeTopZ = TopHit.ImpactPoint.Z;
+	const float LedgeHeight = LedgeTopZ - Feet;
+	const bool bHeightOk = bFromGround
+		? GBMantle::CanMantleFromGround(LedgeHeight, JumpRiseCm, Params)
+		: GBMantle::CanMantleFromAir(LedgeHeight, Params);
+	if (!bHeightOk)
+	{
+		return false;
+	}
+
+	// ---- Room: the capsule must fit at the end and above the start (head room). ----
+	const FVector2D EndXY = Edge2D - WallNormal2D * (Radius + TopInset);
+	const FVector End(EndXY.X, EndXY.Y, LedgeTopZ + HalfHeight + 2.f);
+	const FVector Mid(ActorLocation.X, ActorLocation.Y, End.Z);
+	const FCollisionShape FitShape = FCollisionShape::MakeCapsule(Radius, HalfHeight);
+	if (World->OverlapBlockingTestByChannel(End, FQuat::Identity, ECC_Pawn, FitShape, QueryParams, ResponseParams))
+	{
+		return false;
+	}
+	if (World->OverlapBlockingTestByChannel(Mid, FQuat::Identity, ECC_Pawn, FitShape, QueryParams, ResponseParams))
+	{
+		return false;
+	}
+
+	Out.Start = ActorLocation;
+	Out.End = End;
+	Out.LedgePoint = FVector(Edge2D.X, Edge2D.Y, LedgeTopZ);
+	Out.RiseCm = End.Z - ActorLocation.Z;
+	Out.WallNormal2D = WallNormal2D;
+	return true;
+}

+		if (bFromGround)

+void AGBPlayerCharacter::StartMantle(const FGBMantleLedge& Ledge)
+{
+	ReleaseHeldInputs();
+	if (UCharacterMovementComponent* MoveComp = GetCharacterMovement())
+	{
+		MoveComp->StopMovementImmediately();
