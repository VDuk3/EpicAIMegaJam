void ACharacter::CheckJumpInput(float DeltaTime)
{
	JumpCurrentCountPreJump = JumpCurrentCount;
System.Management.Automation.RemoteException
	if (CharacterMovement)
	{
		if (bPressedJump)
		{
			// If this is the first jump and we're already falling, normally we deny that first jump from happening (by skipping right to the second jump, if available)
			// One exception is if our subclass overrides this to allow first jump to still happen while falling under circumstances they care about
			const bool bFirstJump = JumpCurrentCount == 0;
			if (bFirstJump && CharacterMovement->IsFalling() && !CanJumpWhileFalling())
			{
				JumpCurrentCount++;
			}
System.Management.Automation.RemoteException
			const bool bDidJump = CanJump() && (CharacterCVars::UseLegacyDoJump)? CharacterMovement->DoJump(bClientUpdating) : CharacterMovement->DoJump(bClientUpdating, DeltaTime);
			if (bDidJump)
			{
				// Transition from not (actively) jumping to jumping.
				if (!bWasJumping)
				{
					JumpCurrentCount++;
					JumpForceTimeRemaining = GetJumpMaxHoldTime();
					OnJumped();
				}
			}
System.Management.Automation.RemoteException
			bWasJumping = bDidJump;
		}
	}
}

void ACharacter::ClearJumpInput(float DeltaTime)
{
	if (bPressedJump)
	{
		JumpKeyHoldTime += DeltaTime;
System.Management.Automation.RemoteException
		// Don't disable bPressedJump right away if it's still held.
		// Don't modify JumpForceTimeRemaining because a frame of update may be remaining.
		if (JumpKeyHoldTime >= GetJumpMaxHoldTime())
		{
			bPressedJump = false;
		}
	}
	else
	{
		JumpForceTimeRemaining = 0.0f;
		bWasJumping = false;
	}
}

+bool UGBCombatLibrary::ApplyWeakPointBonus(const AActor* Target, FGBDamageInfo& Info, float WeakPointMult)
+{
+	Info.bHeadshot = false;
+	if (WeakPointMult <= 1.f)
+	{
+		return false;
+	}
+	const AGBEnemyCharacter* Enemy = Cast<AGBEnemyCharacter>(Target);
+	if (!Enemy || !Enemy->IsHeadshotLocation(Info.HitLocation))
+	{
+		return false;
+	}
+	Info.Amount *= WeakPointMult;
+	Info.bHeadshot = true;
+	return true;
+}

+{

+	FMath::SegmentDistToSegmentSafe(AimOrigin, AimEnd, BodyBottom, BodyTop, OutAimPoint, OutBodyPoint);
+	return OutBodyPoint;
